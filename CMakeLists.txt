# --- Toolchain floor guard --------------------------------------------------
# Compiler-floor requirement per PROJECT.md constraints: GCC >= 12, Clang >= 15,
# AppleClang (Xcode 15+, reports as AppleClang >= 15), MSVC v143 (>= 19.30).
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 12)
  message(FATAL_ERROR "mediadiff requires GCC >= 12, found ${CMAKE_CXX_COMPILER_VERSION}")
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 15)
  message(FATAL_ERROR "mediadiff requires Clang >= 15, found ${CMAKE_CXX_COMPILER_VERSION}")
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 15)
  message(FATAL_ERROR "mediadiff requires AppleClang (Xcode 15+) >= 15, found ${CMAKE_CXX_COMPILER_VERSION}")
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 19.30)
  message(FATAL_ERROR "mediadiff requires MSVC v143 (>= 19.30), found ${CMAKE_CXX_COMPILER_VERSION}")
endif()

option(MEDIADIFF_WITH_VMAF "Enable libvmaf-backed quality metrics" OFF)

# --- Configure-time GPL fast-fail (belt-and-braces; the mandatory check is
#     the runtime avutil_license()/avcodec_license()/avformat_license()
#     assertion in tests/unit/test_license.cpp, per D-03) --------------------
# vcpkg manifest-mode installs do not reliably produce the classic dpkg-style
# `installed/vcpkg/status` file, so this scans the resolved install tree for
# GPL-only codec artifacts directly instead of parsing any vcpkg-internal state.
foreach(forbidden_lib x264 x265 xvidcore)
  find_library(_FORBIDDEN_${forbidden_lib} NAMES ${forbidden_lib}
               PATHS "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/lib" NO_DEFAULT_PATH)
  if(_FORBIDDEN_${forbidden_lib})
    message(FATAL_ERROR "GPL-only codec '${forbidden_lib}' resolved into the vcpkg install tree — check vcpkg.json features/overrides.")
  endif()
endforeach()

# --- Check registry generator (D-01, D-05) -----------------------------------
# tomllib is standard-library only from Python 3.11 onward — enforced here
# at configure time (not only inside tools/gen_registry.py itself) so a
# too-old interpreter fails the build immediately with CMake's own
# diagnostic rather than a cryptic Python ImportError three layers down.
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)

set(MEDIADIFF_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated")
set(MEDIADIFF_CHECKS_DEF "${CMAKE_SOURCE_DIR}/src/core/checks.def")

# CONFIGURE_DEPENDS so an added/removed docs/checks/<id>.md file triggers a
# reconfigure on the next build — a plain file(GLOB) result is cached and
# would NOT notice a new file on its own (a known CMake footgun).
file(GLOB MEDIADIFF_CHECK_DOCS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/docs/checks/*.md")

add_custom_command(
  OUTPUT
    "${MEDIADIFF_GENERATED_DIR}/core/check_id.h"
    "${MEDIADIFF_GENERATED_DIR}/core/check_registry.cpp"
    "${MEDIADIFF_GENERATED_DIR}/core/check_explain.cpp"
    "${MEDIADIFF_GENERATED_DIR}/core/checks_manifest.json"
  COMMAND Python3::Interpreter
          "${CMAKE_SOURCE_DIR}/tools/gen_registry.py"
          --checks "${MEDIADIFF_CHECKS_DEF}"
          --docs-dir "${CMAKE_SOURCE_DIR}/docs/checks"
          --out-dir "${MEDIADIFF_GENERATED_DIR}/core"
  DEPENDS
    "${MEDIADIFF_CHECKS_DEF}"
    "${CMAKE_SOURCE_DIR}/tools/gen_registry.py"
    ${MEDIADIFF_CHECK_DOCS}
  COMMENT "Generating check registry from checks.def"
  VERBATIM
)
add_custom_target(generate_registry DEPENDS
  "${MEDIADIFF_GENERATED_DIR}/core/check_id.h"
  "${MEDIADIFF_GENERATED_DIR}/core/check_registry.cpp"
  "${MEDIADIFF_GENERATED_DIR}/core/check_explain.cpp"
  "${MEDIADIFF_GENERATED_DIR}/core/checks_manifest.json"
)

# --- Test-only check registry (02-03-PLAN.md Task 1, D-11) -------------------
# A second, independent registry generated by the SAME script (--symbol-prefix
# test_) from tests/support/test_checks.def, covering all seven comparison
# semantics so the fail-first coverage gate (Task 2) and every later phase's
# comparator tests have real, dispatchable checks to test against without a
# single synthetic id ever entering the production registry above. Linked
# only into the two test executables (tests/unit/CMakeLists.txt,
# tests/integration/CMakeLists.txt) -- never into libmediadiff or mediadiff,
# so `list-checks` stays honest (an acceptance criterion greps the shipped
# binary for `t.` identifiers and requires zero matches).
set(MEDIADIFF_TEST_GENERATED_DIR "${MEDIADIFF_GENERATED_DIR}/test")
set(MEDIADIFF_TEST_CHECKS_DEF "${CMAKE_SOURCE_DIR}/tests/support/test_checks.def")
file(GLOB MEDIADIFF_TEST_CHECK_DOCS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/tests/support/docs/*.md")

add_custom_command(
  OUTPUT
    "${MEDIADIFF_TEST_GENERATED_DIR}/test_check_id.h"
    "${MEDIADIFF_TEST_GENERATED_DIR}/test_check_registry.cpp"
    "${MEDIADIFF_TEST_GENERATED_DIR}/test_check_explain.cpp"
    "${MEDIADIFF_TEST_GENERATED_DIR}/checks_manifest.json"
  COMMAND Python3::Interpreter
          "${CMAKE_SOURCE_DIR}/tools/gen_registry.py"
          --checks "${MEDIADIFF_TEST_CHECKS_DEF}"
          --docs-dir "${CMAKE_SOURCE_DIR}/tests/support/docs"
          --out-dir "${MEDIADIFF_TEST_GENERATED_DIR}"
          --symbol-prefix "test_"
  DEPENDS
    "${MEDIADIFF_TEST_CHECKS_DEF}"
    "${CMAKE_SOURCE_DIR}/tools/gen_registry.py"
    ${MEDIADIFF_TEST_CHECK_DOCS}
  COMMENT "Generating test-only check registry from tests/support/test_checks.def"
  VERBATIM
)
add_custom_target(generate_test_registry DEPENDS
  "${MEDIADIFF_TEST_GENERATED_DIR}/test_check_id.h"
  "${MEDIADIFF_TEST_GENERATED_DIR}/test_check_registry.cpp"
  "${MEDIADIFF_TEST_GENERATED_DIR}/test_check_explain.cpp"
  "${MEDIADIFF_TEST_GENERATED_DIR}/checks_manifest.json"
)

# --- Header compilation gate --------------------------------------------------
# A header nothing includes is a header nothing type-checks, and Phase 2
# adds almost entirely headers to core/ — so every new src/core/*.h is
# declared as a verified header set and compiled standalone under the same

# --- Warnings-as-errors on first-party targets only -------------------------
function(mediadiff_apply_warnings target)
  if(MSVC)
    # /utf-8 per PROJECT.md's Windows toolchain constraint: without it MSVC
    # interprets narrow-string universal-character-name escapes (\uXXXX /
    # \UXXXXXXXX, as used by tests/unit/test_fs_utf8.cpp's non-ASCII
    # fixtures) and any non-ASCII source bytes using the ambient code page
    # instead of UTF-8, which would silently corrupt exactly the kind of
    # literal this plan's shim exists to handle correctly (D-04).
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endfunction()

mediadiff_apply_warnings(libmediadiff)
